const styles = {
  primary:
    'bg-navy text-white shadow-card hover:bg-navy-dark hover:shadow-card-hover',
  accent:
    'bg-sun text-navy-dark shadow-sun hover:bg-sun-deep hover:text-white',
  outline:
    'border-2 border-navy/25 text-navy hover:border-navy hover:bg-navy/5',
  // BARU: tombol tanpa latar, untuk aksi kecil (Lihat Semua, Batal)
  ghost: 'text-navy hover:bg-navy/5',
}

// BARU: ukuran opsional. Default 'md' = ukuran lama (px-6 py-3 text-sm)
const sizes = {
  sm: 'px-4 py-2 text-xs',
  md: 'px-6 py-3 text-sm',
  lg: 'px-8 py-4 text-base',
}

export default function Button({
  variant = 'primary',
  size = 'md',
  to,
  className = '',
  children,
  ...rest
}) {
  const cls = `inline-flex items-center justify-center gap-2 rounded-full font-bold transition duration-200 active:translate-y-px disabled:cursor-not-allowed disabled:opacity-50 disabled:shadow-none ${sizes[size]} ${styles[variant]} ${className}`
  if (to) return <Link to={to} className={cls} {...rest}>{children}</Link>
  return <button className={cls} {...rest}>{children}</button>
}

export default function CartModal() {
  const { items, subtotal, inc, dec, remove, cartOpen, setCartOpen } = useCart()
  const close = () => setCartOpen(false)

  // Tekan Esc = tutup keranjang
  useEffect(() => {
    if (!cartOpen) return
    const onKey = (e) => e.key === 'Escape' && setCartOpen(false)
    window.addEventListener('keydown', onKey)
    return () => window.removeEventListener('keydown', onKey)
  }, [cartOpen, setCartOpen])

  // Kunci scroll halaman di belakang saat keranjang terbuka
  useEffect(() => {
    if (!cartOpen) return
    const prev = document.body.style.overflow
    document.body.style.overflow = 'hidden'
    return () => {
      document.body.style.overflow = prev
    }
  }, [cartOpen])

  const handleRemove = (item) => {
    remove(item.key)
    toast(`${item.product.name} dihapus dari keranjang`)
  }

  return (
    <div
      aria-hidden={!cartOpen}
      className={`fixed inset-0 z-[60] ${
        cartOpen ? 'visible' : 'pointer-events-none invisible delay-300 [transition-property:visibility]'
      }`}
    >
      <div
        onClick={close}
        className={`absolute inset-0 bg-navy-dark/50 transition-opacity duration-300 ${cartOpen ? 'opacity-100' : 'opacity-0'}`}
      />
      <aside
        role="dialog"
        aria-modal="true"
        aria-label="Keranjang Belanja"
        className={`absolute right-0 top-0 flex h-full w-full max-w-md flex-col bg-cream shadow-2xl transition-transform duration-300 ${
          cartOpen ? 'translate-x-0' : 'translate-x-full'
        }`}
      >
        <div className="flex items-center justify-between border-b border-navy/10 px-5 py-4">
          <h2 className="font-display text-2xl font-black italic text-navy">Keranjang Belanja</h2>
          <button
            onClick={close}
            aria-label="Tutup keranjang"
            className="rounded-full p-2 text-navy transition-colors hover:bg-navy/5"
          >
            <X size={22} />
          </button>
        </div>

        {items.length === 0 ? (

function slotsFor(outlet, dateISO) {
  if (!outlet) return []
  const now = new Date()
  const isToday = dateISO === toISO(now)
  const nowMin = now.getHours() * 60 + now.getMinutes()
  const out = []
  for (let m = toMin(outlet.buka); m <= toMin(outlet.tutup) - 60; m += 60) {
    if (isToday && m < nowMin + 30) continue

import { useEffect, useRef, useState } from 'react'

// Bungkus elemen apa pun, nanti muncul halus saat di-scroll ke layar.
// Contoh: <Reveal delay={150}> ... </Reveal>
export default function Reveal({ as: Tag = 'div', delay = 0, className = '', children, ...rest }) {
  const ref = useRef(null)
  const [shown, setShown] = useState(false)

  useEffect(() => {
    const el = ref.current
    if (!el) return
    // browser lama tanpa IntersectionObserver: langsung tampilkan
    if (!('IntersectionObserver' in window)) {
      setShown(true)
      return
    }
    const io = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) {
          setShown(true)
          io.disconnect() // cukup sekali, tidak diulang saat scroll balik
        }
      },
      { threshold: 0.15 }
    )
    io.observe(el)
    return () => io.disconnect()
  }, [])

  return (
    <Tag
      ref={ref}
      style={{ transitionDelay: `${delay}ms` }}
      className={`tr-reveal ${shown ? 'is-in' : ''} ${className}`}
      {...rest}
    >
      {children}
    </Tag>
  )
}

import { useEffect, useRef } from 'react'

// Garis kuning tipis di paling atas layar, memanjang sesuai seberapa jauh halaman di-scroll.
export default function ScrollProgress() {
  const bar = useRef(null)
  const { pathname } = useLocation()

  useEffect(() => {
    const update = () => {
      const max = document.documentElement.scrollHeight - window.innerHeight
      const p = max > 0 ? Math.min(1, window.scrollY / max) : 0
      if (bar.current) bar.current.style.transform = `scaleX(${p})`
    }
    update()
    // tinggi halaman berubah setelah pindah route atau data selesai dimuat
    const t = setTimeout(update, 120)
    window.addEventListener('scroll', update, { passive: true })
    window.addEventListener('resize', update)
    return () => {
      clearTimeout(t)
      window.removeEventListener('scroll', update)
      window.removeEventListener('resize', update)
    }
  }, [pathname])

  return (
    <div
      ref={bar}
      aria-hidden="true"
      style={{ transform: 'scaleX(0)' }}
      className="pointer-events-none fixed left-0 top-0 z-[70] h-[3px] w-full origin-left bg-sun"
    />
  )
}

  const [cartOpen, setCartOpen] = useState(false)

      if (found) return prev.map((i) => (i.key === k ? { ...i, qty: i.qty + qty } : i))
      return [...prev, { key: k, product, variant, qty }]
    })
  }

  const remove = (k) => setItems((p) => p.filter((i) => i.key !== k))
  const clear = () => setItems([])

  // pemanggilan lama placeOrder(total) tetap jalan.

      items: items.map((i) => `${i.product.name} x${i.qty}`),
      customer,
    }

    clear()

  }

    >
      {children}
    </CartContext.Provider>
  )

const wait = (ms) => new Promise((r) => setTimeout(r, ms))

export default function useProducts() {
  const [products, setProducts] = useState([])
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState(null)
  const [attempt, setAttempt] = useState(0)

    const controller = new AbortController()

    async function load() {
      setLoading(true)
      setError(null)
      try {
        const [res] = await Promise.all([
          fetch(API_URL, { signal: controller.signal }),
          wait(MIN_LOADING_MS),
        ])
        if (!res.ok) throw new Error(`HTTP ${res.status}`)
        const data = await res.json()
        if (!Array.isArray(data)) throw new Error('Format data tidak valid')
        setProducts(data)
      } catch (err) {
        if (err.name === 'AbortError') return
        setError('Gagal memuat data. Coba lagi')
      } finally {
        if (!controller.signal.aborted) setLoading(false)
      }
    }

    load()
    return () => controller.abort()
  }, [attempt])

  const reload = useCallback(() => setAttempt((a) => a + 1), [])

  return { products, loading, error, reload }
